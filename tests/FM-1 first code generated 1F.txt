#pragma once
#include <cmath>
#include <cstdint>
#include <algorithm>

#define SAMPLE_RATE 44100
#define TWO_PI 6.28318530717958647692f
#define MAX_VOICES 4 // Capped at 4 to maximize effects memory and CPU allocation

struct SynthParameters {
    float cutoff = 0.4f;
    float resonance = 0.1f;
    float modulationAmount = 0.2f;
    float attack = 0.01f;
    float release = 0.6f;
    
    // Mega Effects Parameters
    float delayTime = 0.5f;
    float delayFeedback = 0.7f; 
    float delayDamp = 0.3f;     
    float delayWetMix = 0.4f;
    
    float reverbSize = 0.85f;    // Enhanced maximum depth
    float reverbDecay = 0.92f;   // Extended long tail scaling
    float reverbWetMix = 0.5f;
};

// ==========================================
// SYNTH VOICE DEFINITION (4-Voice Mode)
// ==========================================
class SynthVoice {
public:
    float phaseCarrier = 0.0f;
    float phaseModulator = 0.0f;
    float carrierFreq = 0.0f;
    float modulatorFreq = 0.0f;
    float envelopeValue = 0.0f;
    bool gateActive = false;
    uint8_t currentNote = 0;
    uint32_t age = 0;

    void triggerNote(uint8_t note, float freq) {
        currentNote = note;
        carrierFreq = freq;
        modulatorFreq = freq * 2.0f;
        gateActive = true;
        envelopeValue = 1.0f;
        age = 0;
    }

    void releaseNote() { gateActive = false; }
    bool isActive() const { return (gateActive || envelopeValue > 0.001f); }

    float processSample(const SynthParameters& params) {
        if (!isActive()) return 0.0f;
        age++;

        phaseModulator += modulatorFreq / SAMPLE_RATE;
        if (phaseModulator >= 1.0f) phaseModulator -= 1.0f;
        float modSignal = std::sin(phaseModulator * TWO_PI) * params.modulationAmount;

        phaseCarrier += (carrierFreq / SAMPLE_RATE) + (modSignal * 0.01f);
        if (phaseCarrier >= 1.0f) phaseCarrier -= 1.0f;
        float rawSample = std::sin(phaseCarrier * TWO_PI);

        if (!gateActive) {
            envelopeValue -= (1.0f / (params.release * SAMPLE_RATE));
            if (envelopeValue < 0.0f) envelopeValue = 0.0f;
        }
        return rawSample * envelopeValue;
    }
};

// ==========================================
// EXTREME DELAY (With Dampening & Tape Saturation)
// ==========================================
class ExtremeDelay {
private:
    static const uint32_t DELAY_BUF_SIZE = 32000; // ~0.72 seconds max
    float bufferL[DELAY_BUF_SIZE] = {0.0f};
    float bufferR[DELAY_BUF_SIZE] = {0.0f};
    uint32_t writePtr = 0;
    float lastOutL = 0.0f;
    float lastOutR = 0.0f;

public:
    void process(float& sampleL, float& sampleR, const SynthParameters& params) {
        uint32_t delaySamples = static_cast<uint32_t>(params.delayTime * (DELAY_BUF_SIZE - 200)) + 100;
        
        int32_t readPtr = static_cast<int32_t>(writePtr) - delaySamples;
        if (readPtr < 0) readPtr += DELAY_BUF_SIZE;

        float wetL = bufferL[readPtr];
        float wetR = bufferR[readPtr];

        float feedL = sampleL + (wetR * params.delayFeedback);
        float feedR = sampleR + (wetL * params.delayFeedback);
        
        // Soft clipping tape distortion simulator
        feedL = std::tanhf(feedL * 1.1f); 
        feedR = std::tanhf(feedR * 1.1f);

        // HF damping loop
        lastOutL += (1.0f - params.delayDamp) * (feedL - lastOutL);
        lastOutR += (1.0f - params.delayDamp) * (feedR - lastOutR);

        bufferL[writePtr] = lastOutL;
        bufferR[writePtr] = lastOutR;
        writePtr = (writePtr + 1) % DELAY_BUF_SIZE;

        sampleL = (sampleL * (1.0f - params.delayWetMix)) + (wetL * params.delayWetMix);
        sampleR = (sampleR * (1.0f - params.delayWetMix)) + (wetR * params.delayWetMix);
    }
};

// ==========================================
// DEEP SPACE REVERB (Expanded 6-Comb Architecture)
// ==========================================
class ExtremeReverb {
private:
    // Expanded to 6 parallel combs for incredible acoustic density
    static const int NUM_COMBS = 6;
    const int combTimes[NUM_COMBS] = {1117, 1373, 1583, 1721, 1907, 2111};
    float combBuffers[NUM_COMBS][2200] = {{0.0f}}; 
    int combIndices[NUM_COMBS] = {0};

    // Upgraded double-stage stereo diffuser all-pass networks
    float apBufferL1[721] = {0.0f}, apBufferL2[443] = {0.0f};
    float apBufferR1[877] = {0.0f}, apBufferR2[523] = {0.0f};
    int apIdxL1 = 0, apIdxL2 = 0;
    int apIdxR1 = 0, apIdxR2 = 0;

public:
    void process(float& sampleL, float& sampleR, const SynthParameters& params) {
        float inputSum = (sampleL + sampleR) * 0.5f;
        float combOutSum = 0.0f;

        for (int i = 0; i < NUM_COMBS; ++i) {
            int bufSize = static_cast<int>(combTimes[i] * params.reverbSize);
            if (bufSize < 10) bufSize = 10;

            float out = combBuffers[i][combIndices[i]];
            combBuffers[i][combIndices[i]] = inputSum + (out * params.reverbDecay);
            
            combIndices[i] = (combIndices[i] + 1) % bufSize;
            combOutSum += out;
        }
        combOutSum *= (1.0f / static_cast<float>(NUM_COMBS));

        // Left Channel Diffusion (Cascaded Stage 1 -> Stage 2)
        float apInL = combOutSum + sampleL;
        float apOutL1 = -0.6f * apInL + apBufferL1[apIdxL1];
        apBufferL1[apIdxL1] = apInL + (apOutL1 * 0.6f);
        apIdxL1 = (apIdxL1 + 1) % 721;

        float apOutL2 = -0.5f * apOutL1 + apBufferL2[apIdxL2];
        apBufferL2[apIdxL2] = apOutL1 + (apOutL2 * 0.5f);
        apIdxL2 = (apIdxL2 + 1) % 443;

        // Right Channel Diffusion (Cascaded Stage 1 -> Stage 2)
        float apInR = combOutSum + sampleR;
        float apOutR1 = -0.6f * apInR + apBufferR1[apIdxR1];
        apBufferR1[apIdxR1] = apInR + (apOutR1 * 0.6f);
        apIdxR1 = (apIdxR1 + 1) % 877;

        float apOutR2 = -0.5f * apOutR1 + apBufferR2[apIdxR2];
        apBufferR2[apIdxR2] = apOutR1 + (apOutR2 * 0.5f);
        apIdxR2 = (apIdxR2 + 1) % 523;

        // Wet/Dry mix output mapping
        sampleL = (sampleL * (1.0f - params.reverbWetMix)) + (apOutL2 * params.reverbWetMix);
        sampleR = (sampleR * (1.0f - params.reverbWetMix)) + (apOutR2 * params.reverbWetMix);
    }
};

// ==========================================
// CORE AUDIO AUDIO ENGINE MODULE
// ==========================================
class PolyCustomEngine {
private:
    SynthVoice voices[MAX_VOICES];
    SynthParameters params;
    ExtremeDelay delayEffect;
    ExtremeReverb reverbEffect;
    float lpL = 0.0f, lpR = 0.0f;

public:
    PolyCustomEngine() {}

    void noteOn(uint8_t note, uint8_t velocity) {
        float frequency = 440.0f * std::pow(2.0f, (note - 69) / 12.0f);
        for (int i = 0; i < MAX_VOICES; ++i) {
            if (voices[i].isActive() && voices[i].currentNote == note) {
                voices[i].triggerNote(note, frequency);
                return;
            }
        }
        for (int i = 0; i < MAX_VOICES; ++i) {
            if (!voices[i].isActive()) {
                voices[i].triggerNote(note, frequency);
                return;
            }
        }
        int oldestIdx = 0; uint32_t maxAge = 0;
        for (int i = 0; i < MAX_VOICES; ++i) {
            if (voices[i].age > maxAge) { maxAge = voices[i].age; oldestIdx = i; }
        }
        voices[oldestIdx].triggerNote(note, frequency);
    }

    void noteOff(uint8_t note) {
        for (int i = 0; i < MAX_VOICES; ++i) {
            if (voices[i].currentNote == note) voices[i].releaseNote();
        }
    }

    void updateParameter(uint8_t paramId, float value) {
        switch(paramId) {
            case 10: params.cutoff = value; break;
            case 11: params.resonance = value; break;
            case 12: params.modulationAmount = value * 2.5f; break;
            case 13: params.attack = (value * 2.0f) + 0.001f; break;
            case 14: params.release = (value * 5.0f) + 0.01f; break;
            
            // Extreme Delay
            case 20: params.delayTime = 0.1f + (value * 0.9f); break;
            case 21: params.delayFeedback = value * 0.98f; break; 
            case 22: params.delayDamp = value * 0.8f; break;
            case 23: params.delayWetMix = value; break;

            // Extreme Reverb
            case 24: params.reverbSize = 0.2f + (value * 0.8f); break;
            case 25: params.reverbDecay = 0.5f + (value * 0.49f); // Pushed right up to the edge of self-oscillation
            case 26: params.reverbWetMix = value; break;
        }
    }

    void processAudioBuffer(float* outputBuffer, uint32_t numSamples) {
        for (uint32_t i = 0; i < numSamples; ++i) {
            float mixedSample = 0.0f;
            for (int v = 0; v < MAX_VOICES; ++v) {
                mixedSample += voices[v].processSample(params);
            }
            // Dynamic gain normalization adjusted for 4-voice blending bounds
            mixedSample *= (1.0f / static_cast<float>(MAX_VOICES)) * 1.4f;

            lpL += (params.cutoff * 0.15f) * (mixedSample - lpL);
            lpR += (params.cutoff * 0.15f) * (mixedSample - lpR);

            float outL = lpL;
            float outR = lpR;

            delayEffect.process(outL, outR, params);
            reverbEffect.process(outL, outR, params);

            // Hard clipping safety rail protection
            outputBuffer[i * 2]     = std::max(-1.0f, std::min(1.0f, outL));
            outputBuffer[i * 2 + 1] = std::max(-1.0f, std::min(1.0f, outR));
        }
    }
};
